function(conduit_add_example target src)
    add_executable(${target} ${src})
    target_link_libraries(${target} PRIVATE conduit::conduit)
    conduit_set_warnings(${target} PRIVATE)
    conduit_enable_sanitizers(${target})
    conduit_enable_clang_tidy(${target})
endfunction()

conduit_add_example(conduit_hello                    hello.cpp)

conduit_add_example(conduit_pattern_listener         pattern_listener.cpp)

conduit_add_example(conduit_serialization_roundtrip  serialization_roundtrip.cpp)
